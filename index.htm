function calc(){

var salary=parseFloat(document.getElementById('salary').value)||0;

var freq=parseInt(document.getElementById('freq').value)||26;

var status=document.getElementById('status').value;

 

var gross=salary/freq;

 

var fedTax=0;

if(status==='single'){

if(salary<=11925)fedTax=salary*0.10;

else if(salary<=48475)fedTax=1192.5+(salary-11925)*0.12;

else if(salary<=103350)fedTax=5578.5+(salary-48475)*0.22;

else fedTax=17639+(salary-103350)*0.24;

}else{

if(salary<=23850)fedTax=salary*0.10;

else if(salary<=96950)fedTax=2385+(salary-23850)*0.12;

else if(salary<=206700)fedTax=11157+(salary-96950)*0.22;

else fedTax=35292+(salary-206700)*0.24;

}

var fedPer=fedTax/freq;

 

var dcTax=0;

if(salary<=10000)dcTax=salary*0.04;

else if(salary<=40000)dcTax=400+(salary-10000)*0.06;

else if(salary<=60000)dcTax=2200+(salary-40000)*0.065;

else if(salary<=350000)dcTax=3500+(salary-60000)*0.085;

else dcTax=28150+(salary-350000)*0.0925;

var dcPer=dcTax/freq;

 

var ss=Math.min(salary,176100)*0.062;

var medicare=salary*0.0145;

var ficaPer=(ss+medicare)/freq;

 

var net=gross-fedPer-dcPer-ficaPer;

 

document.getElementById('result').innerHTML='<div class="big">$'+net.toFixed(2)+' per paycheck</div>'+

'<div style="margin-top:10px;font-size:15px">Gross: $'+gross.toFixed(2)+'<br>'+

'Federal Tax: -$'+fedPer.toFixed(2)+'<br>'+

'DC Tax: -$'+dcPer.toFixed(2)+'<br>'+

'FICA: -$'+ficaPer.toFixed(2)+'<br>'+

'<strong>Net: $'+net.toFixed(2)+'</strong></div>';

}
